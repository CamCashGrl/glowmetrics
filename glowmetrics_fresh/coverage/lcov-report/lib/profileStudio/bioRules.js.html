// ============================================================
// CREATOR PROFILE STUDIO — BIO SAFETY & VALIDATION RULES
// Server-side only. The AI prompt already instructs the model to follow
// these rules, but nothing generated by an LLM is trusted blindly — every
// bio is re-checked here before it's ever returned to a client.
// ============================================================
 
import { z } from 'zod'
import { ALL_PLATFORMS, PERSONA_TAGS, TONE_OPTIONS, SERVICE_OPTIONS, PRIMARY_CTA_OPTIONS, BIO_FORMATS, BIO_LENGTHS, EMOJI_LEVELS, PLATFORM_CHAR_CAPS } from './constants'
 
const BIO_FORMAT_IDS = BIO_FORMATS.map(f => f.id)
const BIO_LENGTH_IDS = BIO_LENGTHS.map(l => l.id)
const EMOJI_LEVEL_IDS = EMOJI_LEVELS.map(e => e.id)
 
const shortText = (max) => z.string().trim().max(max).optional().default('')
 
export const BioIdentitySchema = z.object({
  displayName: z.string().trim().min(1).max(60),
  username: z.string().trim().max(60).optional().default(''),
  pronouns: shortText(30),
  ageDisplay: z.enum(['none', '18plus', 'custom']).default('none'),
  ageDisplayValue: z.string().trim().max(20).optional().default(''),
  locationGeneral: shortText(80),
  languages: z.array(z.string().trim().max(30)).max(10).default([]),
  experienceLevel: z.enum(['new', 'growing', 'established']).default('new'),
  platforms: z.array(z.enum(ALL_PLATFORMS)).max(ALL_PLATFORMS.length).default([]),
})
 
export const BioPersonaSchema = z.object({
  personaTags: z.array(z.string().max(40)).max(30).default([]),
  personalityWords: z.array(z.string().trim().max(24)).max(3).default([]),
  desiredAudience: shortText(200),
  brandColors: z.array(z.string().trim().max(20)).max(6).default([]),
  signaturePhrase: shortText(120),
  roomDifference: shortText(300),
  strongestQualities: shortText(300),
  tone: z.array(z.enum(TONE_OPTIONS)).max(TONE_OPTIONS.length).default([]),
})
 
export const BioBoundariesSchema = z.object({
  hardNo: z.array(z.string().trim().max(80)).max(20).default([]),
  topics: shortText(400),
  mentionPublicly: z.boolean().default(false),
  softPhrase: z.boolean().default(false),
  ageReminder: z.boolean().default(false),
})
 
export const BioContentSchema = z.object({
  services: z.array(z.enum(SERVICE_OPTIONS)).max(SERVICE_OPTIONS.length).default([]),
  boundaries: BioBoundariesSchema.default({}),
})
 
export const BioScheduleSchema = z.object({
  days: z.array(z.string().max(12)).max(7).default([]),
  hours: shortText(60),
  timezone: shortText(60),
  consistent: z.boolean().default(true),
  // The form sends '' (not a missing key) when no CTA is picked yet —
  // normalize that to undefined before the enum check so an untouched
  // "optional" field doesn't fail validation.
  ctaPrimary: z.preprocess(v => (v === '' ? undefined : v), z.enum(PRIMARY_CTA_OPTIONS).optional()),
  ctaSecondary: z.preprocess(v => (v === '' ? undefined : v), z.enum(PRIMARY_CTA_OPTIONS).optional()),
  socialUsernames: z.record(z.string(), z.string().max(60)).optional().default({}),
})
 
export const BioGenerateRequestSchema = z.object({
  identity: BioIdentitySchema,
  persona: BioPersonaSchema,
  content: BioContentSchema,
  schedule: BioScheduleSchema,
  format: z.enum(BIO_FORMAT_IDS),
  length: z.enum(BIO_LENGTH_IDS).default('medium'),
  emoji: z.enum(EMOJI_LEVEL_IDS).default('minimal'),
})
 
export const BioImproveRequestSchema = z.object({
  existingBio: z.string().trim().min(1).max(4000),
  format: z.enum(BIO_FORMAT_IDS).default('full_cam_profile'),
})
 
// ── Hard safety filters ──────────────────────────────────────────────────
// Any match here means the bio is rejected outright, not "cleaned up" —
// these are non-negotiable per the age-safety requirements.
const UNDERAGE_PATTERNS = [
  /\bunder\s?-?\s?18\b/i, /\bminor(s)?\b/i, /\bteen(s|ager|agers)?\b/i,
  /\bschool\s?girl\b/i, /\bhigh\s?school\b/i, /\bmiddle\s?school\b/i,
  /\bfreshman\b/i, /\bjunior\s?high\b/i, /\b1[0-7]\s?(years?\s?old|y\/?o)\b/i,
  /\bunderage\b/i, /\bbarely\s?legal\b/i,
]
 
// Unsupported/overclaiming language — stripped defensively even though the
// prompt already forbids it.
const BANNED_SUPERLATIVES = [
  /\btop\s+creator\b/gi, /\bnumber\s+one\b/gi, /\b#\s?1\b/gi,
  /\bbest\s+(cam\s+)?(girl|model|creator)\s+(on|in)\b/gi,
  /\bmost\s+popular\b/gi, /\bworld'?s\s+best\b/gi,
]
 
// Restricted/explicit platform terms that should be swapped for a safer
// equivalent when a platform-neutral phrase works just as well.
const RESTRICTED_PHRASE_SWAPS = [
  [/\bfull\s+nude(ity)?\b/gi, 'uncensored content'],
  [/\bhardcore\b/gi, 'explicit content'],
]
 
function containsAny(text, patterns) {
  return patterns.some(p => p.test(text))
}
 
function stripPatterns(text, patterns) {
  let out = text
  for (const p of patterns) out = out.replace(p, '')
  return out.replace(/\s{2,}/g, ' ').replace(/\s+([,.!?])/g, '$1').trim()
}
 
// Keyword fingerprints used to catch a bio that mentions a service the
// creator did NOT select. Best-effort defensive net on top of the prompt
// constraint — not a substitute for it.
const SERVICE_KEYWORDS = {
  'Private shows': [/\bprivate show/i],
  'Exclusive private shows': [/\bexclusive private/i],
  'Games': [/\bgames?\b/i],
  'Tip goals': [/\btip goal/i],
  'Lovense or interactive toys': [/\blovense\b/i, /\binteractive toy/i, /\bohmibod\b/i],
  'Roleplay': [/\broleplay/i],
  'Cosplay': [/\bcosplay/i],
  'Dancing': [/\bdanc(e|ing)\b/i],
  'Sexting': [/\bsexting\b/i],
  'Custom content': [/\bcustom (content|video|clip)/i],
  'Fan club': [/\bfan club\b/i],
}
 
/**
 * Validates a generated bio against the creator's actual selections.
 * Returns { text, warnings, blocked } — blocked=true means this bio must
 * not be shown (hard safety failure), warnings are non-fatal notices shown
 * to the creator so they can review before publishing.
 */
export function sanitizeGeneratedBio(rawText, { services = [], maxChars } = {}) {
  const warnings = []
 
  if (typeof rawText !== 'string' || !rawText.trim()) {
    return { text: '', warnings, blocked: true, blockReason: 'empty_generation' }
  }
 
  if (containsAny(rawText, UNDERAGE_PATTERNS)) {
    return { text: '', warnings, blocked: true, blockReason: 'underage_language' }
  }
 
  let text = stripPatterns(rawText, BANNED_SUPERLATIVES)
  if (text !== rawText.replace(/\s{2,}/g, ' ').trim()) {
    warnings.push('Removed an unsupported claim (e.g. "top creator") from the generated text.')
  }
 
  for (const [pattern, replacement] of RESTRICTED_PHRASE_SWAPS) {
    if (pattern.test(text)) {
      text = text.replace(pattern, replacement)
      warnings.push('Swapped in platform-safer wording for a restricted term.')
    }
  }
 
  const selectedSet = new Set(services)
  for (const [service, patterns] of Object.entries(SERVICE_KEYWORDS)) {
    if (selectedSet.has(service)) continue
    if (patterns.some(p => p.test(text))) {
      warnings.push(`This bio may reference "${service}", which wasn't selected in your services — please review before publishing.`)
    }
  }
 
  if (maxChars && text.length > maxChars) {
    warnings.push(`Trimmed to fit the ${maxChars}-character platform limit.`)
    text = text.slice(0, maxChars).replace(/\s+\S*$/, '').trim()
  }
 
  return { text, warnings, blocked: false }
}
 
export function charCap(formatId, lengthId) {
  const formatCap = PLATFORM_CHAR_CAPS[formatId]
  const lengthCap = BIO_LENGTHS.find(l => l.id === lengthId)?.maxChars
  if (formatCap && lengthCap) return Math.min(formatCap, lengthCap)
  return formatCap || lengthCap || 500
}
