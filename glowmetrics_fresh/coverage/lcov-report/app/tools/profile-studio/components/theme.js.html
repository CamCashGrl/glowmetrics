// Collapsible section for supporting/marketing content — keeps the page
// short with everything closed, while each section's content (built by the
// existing marketing components) is unchanged, just wrapped.
export function Accordion({ title, icon, subtitle, defaultOpen = false, children, onToggle }) {
  const [open, setOpen] = React.useState(defaultOpen)
  const toggle = () => {
    const next = !open
    setOpen(next)
    onToggle?.(next)
  }
  return (
    <div style={{ background: T.card, border: `1px solid ${T.border}`, borderRadius: 16, marginBottom: 12, overflow: 'hidden' }}>
      <button
        onClick={toggle}
        className="ps-btn"
        aria-expanded={open}
        style={{ width: '100%', textAlign: 'left', padding: '18px 22px', background: 'none', border: 'none', cursor: 'pointer', display: 'flex', justifyContent: 'space-between', alignItems: 'center', gap: 12 }}
      >
        <span style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
          {icon && <span style={{ fontSize: 18 }}>{icon}</span>}
          <span>
            <span style={{ fontSize: 15.5, fontWeight: 800, color: T.white, display: 'block' }}>{title}</span>
            {subtitle && <span style={{ fontSize: 12, color: T.faint, fontWeight: 500 }}>{subtitle}</span>}
          </span>
        </span>
        <span style={{ color: T.cyan, fontSize: 20, flexShrink: 0, transform: open ? 'rotate(45deg)' : 'none', transition: 'transform 0.15s' }}>+</span>
      </button>
      {open && <div style={{ padding: '0 22px 26px' }}>{children}</div>}
    </div>
  )
}

export function PillGroup({ options, value = [], onToggle, max }) {
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8 }}>
      {options.map(opt => {
        const id = typeof opt === 'string' ? opt : opt.id
        const label = typeof opt === 'string' ? opt : opt.label
        const active = value.includes(id)
        const disabled = !active && max && value.length >= max
        return <Pill key={id} active={active} disabled={disabled} onClick={() => onToggle(id)}>{label}</Pill>
      })}
    </div>
  )
}

export function LoadingStaged({ messages, active }) {
  const [idx, setIdx] = useState(0)
  React.useEffect(() => {
    if (!active) { setIdx(0); return }
    const t = setInterval(() => setIdx(i => Math.min(i + 1, messages.length - 1)), 1100)
    return () => clearInterval(t)
  }, [active, messages.length])
  if (!active) return null
  return (
    <div className="ps-fade" style={{ textAlign: 'center', padding: '40px 20px' }}>
      <div style={{ width: 44, height: 44, margin: '0 auto 18px', borderRadius: '50%', border: `3px solid rgba(255,255,255,0.08)`, borderTopColor: T.cyan, animation: 'psSpin 0.8s linear infinite' }} />
      <div style={{ fontSize: 15, fontWeight: 700, color: T.white, animation: 'psPulse 1.4s ease infinite' }}>{messages[idx]}</div>
      <div style={{ marginTop: 14, height: 6, borderRadius: 999, background: 'rgba(255,255,255,0.08)', maxWidth: 260, margin: '14px auto 0', overflow: 'hidden' }}>
        <div style={{ height: '100%', borderRadius: 999, background: GRADIENT, width: `${((idx + 1) / messages.length) * 100}%`, transition: 'width 0.4s ease' }} />
      </div>
    </div>
  )
}
