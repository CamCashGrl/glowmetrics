import { NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'
import { verifyAdminToken, getAdminTokenFromCookieHeader } from '../../../../lib/adminAuth'
 
const TEST_EMAIL_PATTERNS = [
  'daphneoncam+test',
  'daphenoncam+testing',
  'camcashgrl+testing',
  'daphneoncam@gmail.com',  // founder account — exclude from public metrics
]
 
function isTestAccount(email) {
  if (!email) return false
  const e = email.toLowerCase()
  return TEST_EMAIL_PATTERNS.some(p => e.includes(p))
}
 
function sb() {
  return createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.SUPABASE_SERVICE_ROLE_KEY
  )
}
 
function getDashboardStage(tipCount, totalUsd, liveActive) {
  if (liveActive) return 'live-active'
  if (tipCount === 0) return 'no-data'
  if (tipCount >= 250 && totalUsd >= 100) return 'full-dashboard'
  if (tipCount >= 100 && totalUsd >= 50) return 'growing-insights'
  if (tipCount >= 25 && totalUsd >= 10) return 'first-valid-data'
  return 'tiny-import'
}
 
function getOnboardingStage(beta, uploadCount) {
  if (!beta) return 'no-profile'
  if (beta.onboarding_complete) return 'complete'
  if (uploadCount > 0) return 'imported'
  return 'started'
}
 
export async function GET(req) {
  const token = getAdminTokenFromCookieHeader(req.headers.get('cookie'))
  if (!verifyAdminToken(token)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }
 
  const client = sb()
  const now = new Date()
  const h24 = new Date(now - 86400000).toISOString()
  const h12 = new Date(now - 12 * 3600000).toISOString()
  const d7  = new Date(now - 7  * 86400000).toISOString()
  const d30 = new Date(now - 30 * 86400000).toISOString()
 
  const [
    authResult,
    profilesResult,
    betaResult,
    uploadsResult,
    tipsResult,
    liveTipsResult,
    subsResult,
    feedbackResult,
    extensionEventsResult,
  ] = await Promise.all([
    client.auth.admin.listUsers({ perPage: 1000 }),
    client.from('profiles').select('id, created_at, plan'),
    client.from('beta_users').select('user_id, email, onboarding_complete, csv_upload_count, signed_up_at'),
    client.from('csv_uploads').select('user_id, platform, status, total_rows, inserted_rows, total_usd, uploaded_at').order('uploaded_at', { ascending: false }),
    client.from('tips').select('user_id, usd, platform, created_at'),
    client.from('live_tips').select('user_id, received_at, usd').order('received_at', { ascending: false }),
    client.from('subscriptions').select('user_id, status, plan_type, current_period_end'),
    client.from('feedback').select('message, page_url, created_at, user_id').order('created_at', { ascending: false }).limit(10),
    client.from('analytics_events')
      .select('user_id, event_name, created_at')
      .in('event_name', ['extension_installed', 'extension_updated', 'extension_download_clicked', 'extension_active']),
  ])
 
  const allUsers  = authResult.data?.users || []
  const profiles  = profilesResult.data || []
  const betas     = betaResult.data || []
  const uploads   = uploadsResult.data || []
  const tips      = tipsResult.data || []
  const liveTips  = liveTipsResult.data || []
  const subs      = subsResult.data || []
  const feedback  = feedbackResult.data || []
  const extEvents = extensionEventsResult.data || []
 
  const realUsers = allUsers.filter(u => !isTestAccount(u.email))
  const testUsers = allUsers.filter(u => isTestAccount(u.email))
 
  // Index maps
  const profileMap = Object.fromEntries(profiles.map(p => [p.id, p]))
  const betaMap    = Object.fromEntries(betas.map(b => [b.user_id, b]))
  const subMap     = Object.fromEntries(subs.map(s => [s.user_id, s]))
 
  // Per-user tip aggregates (include platform)
  const tipsByUser = {}
  for (const t of tips) {
    if (!tipsByUser[t.user_id]) tipsByUser[t.user_id] = { count: 0, usd: 0, last: null, platforms: new Set() }
    tipsByUser[t.user_id].count++
    tipsByUser[t.user_id].usd += parseFloat(t.usd) || 0
    if (t.platform) tipsByUser[t.user_id].platforms.add(t.platform.toLowerCase())
    if (!tipsByUser[t.user_id].last || t.created_at > tipsByUser[t.user_id].last)
      tipsByUser[t.user_id].last = t.created_at
  }
 
  // Per-user upload aggregates
  const uploadsByUser = {}
  for (const u of uploads) {
    if (!uploadsByUser[u.user_id]) uploadsByUser[u.user_id] = []
    uploadsByUser[u.user_id].push(u)
  }
 
  // Live tips last 12h per user
  const liveActive12h = new Set(liveTips.filter(lt => lt.received_at > h12).map(lt => lt.user_id))
 
  // Platform breakdown from tips table
  const platformBreakdown = {}
  for (const t of tips) {
    const p = (t.platform || 'unknown').toLowerCase()
    platformBreakdown[p] = (platformBreakdown[p] || 0) + 1
  }
 
  // Build per-user rows (real users only)
  const userRows = realUsers.map(u => {
    const profile     = profileMap[u.id]
    const beta        = betaMap[u.id]
    const sub         = subMap[u.id]
    const userUploads = uploadsByUser[u.id] || []
    const tipStat     = tipsByUser[u.id] || { count: 0, usd: 0, last: null, platforms: new Set() }
    const isLive      = liveActive12h.has(u.id)
    const platforms   = [...(tipStat.platforms instanceof Set ? tipStat.platforms : new Set())]
 
    return {
      id:              u.id,
      email:           u.email,
      createdAt:       u.created_at,
      lastSignIn:      u.last_sign_in_at || null,
      plan:            sub?.plan_type || profile?.plan || 'free',
      onboardingStage: getOnboardingStage(beta, userUploads.length),
      dashboardStage:  getDashboardStage(tipStat.count, tipStat.usd, isLive),
      uploadCount:     userUploads.length,
      tipCount:        tipStat.count,
      totalUsd:        +tipStat.usd.toFixed(2),
      liveActive:      isLive,
      subStatus:       sub?.status || 'free',
      platforms,
      lastActivity:    tipStat.last || u.last_sign_in_at || null,
    }
  }).sort((a, b) => (b.createdAt > a.createdAt ? 1 : -1))
 
  // KPIs
  const totalUsers         = allUsers.length
  const totalRealUsers     = realUsers.length
  const newUsers24h        = realUsers.filter(u => u.created_at > h24).length
  const newUsers7d         = realUsers.filter(u => u.created_at > d7).length
  const newUsers30d        = realUsers.filter(u => u.created_at > d30).length
  const onboardingComplete = betas.filter(b => b.onboarding_complete).length
  const usersWithImports   = Object.keys(uploadsByUser).length
  const usersWithTipData   = Object.keys(tipsByUser).length
  const liveModeUsers      = new Set(liveTips.map(lt => lt.user_id)).size
  const activeUsers24h     = new Set([
    ...tips.filter(t => t.created_at > h24).map(t => t.user_id),
    ...liveTips.filter(lt => lt.received_at > h24).map(lt => lt.user_id),
  ]).size
 
  // Onboarding funnel
  const funnel = {
    signedUp:            totalRealUsers,
    hasProfile:          profiles.filter(p => realUsers.some(u => u.id === p.id)).length,
    startedOnboarding:   betas.filter(b => realUsers.some(u => u.id === b.user_id)).length,
    hasImports:          userRows.filter(u => u.uploadCount > 0).length,
    completedOnboarding: betas.filter(b => b.onboarding_complete && realUsers.some(u => u.id === b.user_id)).length,
  }
 
  // Import health
  const uploadPlatformBreakdown = {}
  for (const u of uploads) {
    uploadPlatformBreakdown[u.platform] = (uploadPlatformBreakdown[u.platform] || 0) + 1
  }
  const importHealth = {
    total:   uploads.length,
    failed:  uploads.filter(u => u.status === 'failed').length,
    last24h: uploads.filter(u => u.uploaded_at > h24).length,
    last7d:  uploads.filter(u => u.uploaded_at > d7).length,
    platforms: uploadPlatformBreakdown,
    recent: uploads.slice(0, 8).map(u => ({
      userEmail: allUsers.find(usr => usr.id === u.user_id)?.email || u.user_id,
      platform:  u.platform,
      status:    u.status,
      rows:      u.total_rows,
      inserted:  u.inserted_rows,
      usd:       u.total_usd,
      at:        u.uploaded_at,
    })),
  }
 
  // Live mode health
  const liveMode = {
    totalUsers:    liveModeUsers,
    tips24h:       liveTips.filter(lt => lt.received_at > h24).length,
    tips12h:       liveTips.filter(lt => lt.received_at > h12).length,
    lastTip:       liveTips[0]?.received_at || null,
    currentlyLive: liveActive12h.size,
  }
 
  // Extension health (installs/downloads/usage — see chrome-extension/background.js
  // and app/components/ExtensionDownloadButton.js for where these events fire)
  const extension = {
    installs:         extEvents.filter(e => e.event_name === 'extension_installed').length,
    updates:          extEvents.filter(e => e.event_name === 'extension_updated').length,
    downloadClicks:   extEvents.filter(e => e.event_name === 'extension_download_clicked').length,
    activeUsers:      new Set(extEvents.filter(e => e.event_name === 'extension_active' && e.user_id).map(e => e.user_id)).size,
    active24h:        new Set(extEvents.filter(e => e.event_name === 'extension_active' && e.user_id && e.created_at > h24).map(e => e.user_id)).size,
    lastInstall:      extEvents.filter(e => e.event_name === 'extension_installed').sort((a, b) => b.created_at > a.created_at ? 1 : -1)[0]?.created_at || null,
  }
 
  // Subscription breakdown
  const subBreakdown = { active: 0, trialing: 0, cancelled: 0, free: 0 }
  for (const s of subs) {
    if (s.status === 'active')                                   subBreakdown.active++
    else if (s.status === 'trialing')                            subBreakdown.trialing++
    else if (s.status === 'canceled' || s.status === 'cancelled') subBreakdown.cancelled++
    else                                                         subBreakdown.free++
  }
  subBreakdown.free += Math.max(0, totalRealUsers - subs.length)
 
  // Dashboard stage breakdown (real users only)
  const stageBreakdown = {}
  for (const u of userRows) {
    stageBreakdown[u.dashboardStage] = (stageBreakdown[u.dashboardStage] || 0) + 1
  }
 
  // Signup trend — last 30 days by day (real users)
  const trendMap = {}
  for (const u of realUsers) {
    const day = u.created_at?.slice(0, 10)
    if (day && day >= d30.slice(0, 10)) {
      trendMap[day] = (trendMap[day] || 0) + 1
    }
  }
  const signupsByDay = Object.entries(trendMap)
    .sort((a, b) => a[0].localeCompare(b[0]))
    .map(([date, count]) => ({ date, count }))
 
  return NextResponse.json({
    kpis: {
      totalUsers,
      totalRealUsers,
      testUsers: testUsers.length,
      newUsers24h,
      newUsers7d,
      newUsers30d,
      onboardingComplete,
      usersWithImports,
      usersWithTipData,
      usersWithZeroTips: totalRealUsers - usersWithTipData,
      liveModeUsers,
      activeUsers24h,
      totalTipRows:    tips.length,
      totalLiveTips:   liveTips.length,
      totalCsvUploads: uploads.length,
      betaUsers:       betas.length,
    },
    funnel,
    stageBreakdown,
    platformBreakdown,
    users:        userRows,
    importHealth,
    liveMode,
    extension,
    subscriptions: subBreakdown,
    feedback,
    signupsByDay,
    generatedAt:  now.toISOString(),
  })
}
